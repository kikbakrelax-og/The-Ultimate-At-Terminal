  <div class="ref-body" id="tab-hayes">
    <div class="rh hayes">Hayes AT command set — origin (1977–1981)</div>
    The Hayes Smartmodem command set is the root every subsequent AT dialect grows from.
    All GSM / CDMA / LTE extensions inherit its syntax and session model.<br><br>
    <span class="rk">AT</span> — attention prefix. Modem watches for this constantly.<br>
    <span class="rk">ATZ</span> — reset to stored NVRAM profile. Does <em>not</em> restore factory defaults.<br>
    <span class="rk">AT&amp;F</span> — restore factory defaults. Combine with AT&amp;W to persist.<br>
    <span class="rk">AT&amp;W</span> — write current config to NVRAM profile.<br>
    <span class="rk">ATE0 / ATE1</span> — echo off / on. E0 stops modem echoing commands back — useful for scripting.<br>
    <span class="rk">ATV0 / ATV1</span> — result code format: V0=numeric (0=OK,4=ERROR), V1=verbose text.<br>
    <span class="rk">ATQ0 / ATQ1</span> — Q1 suppresses all result codes.<br>
    <span class="rk">ATX0–X4</span> — extended result code set. X4 enables connect-speed codes.<br>
    <span class="rk">ATH</span> — on-hook (hang up).<br>
    <span class="rk">ATA</span> — answer incoming call.<br>
    <span class="rk">ATO</span> — return to online data mode from command mode.<br>
    <span class="rk">AT+++</span> — escape sequence: pause-+++- pause returns to command mode mid-call.<br>
    <span class="rk">ATD</span> — dial. ATDTnumber (tone), ATDPnumber (pulse), ATDL (redial), ATD; (dial, stay in cmd mode).<br>
    <span class="rk">AT&amp;C1</span> — DCD follows actual carrier state.<br>
    <span class="rk">AT&amp;D2</span> — hang up + return to cmd mode when DTR drops.<br>
    <span class="rk">AT&amp;K3</span> — RTS/CTS hardware flow control.<br>

    <div class="rh hayes">S-registers</div>
    <span class="rk">ATS0=n</span> — auto-answer after n rings. 0 = disabled.<br>
    <span class="rk">ATS2=43</span> — escape character (ASCII 43 = '+').<br>

/* ══════════════════════════════════════════════════════════════
   OUTPUT / appendLine
═══════════════════════════════════════════════════════════════ */
function ts() {
  const d = new Date();
  return d.toTimeString().slice(0,8);
}

/*
  type values:
    sent / sent2          → command sent on P1 / P2
    ok / err / cme        → result codes
    recv / recv2          → unsolicited / response from P1 / P2
    info                  → local status message
    imsi / sw-resp / sw-note / hayes-resp / apdu-resp / etsi-resp / qcom-resp
*/
function appendLine(text, type='recv', port=1) {
  state.lineCount++;
  const row  = document.createElement('div');
  row.className = 'line ' + type;

  const pfxMap = {
    sent:'▶', sent2:'▶', ok:'✓', err:'✗', cme:'⚠',
    info:'·', recv:'◀', recv2:'◀', imsi:'★',
    'sw-resp':'◀','sw-note':'·','hayes-resp':'◀',
    'apdu-resp':'◀','etsi-resp':'◀','qcom-resp':'◀',
  };

  const pfx  = document.createElement('span');
  pfx.className = 'pfx';
  pfx.textContent = pfxMap[type] || '·';

  const tsEl = document.createElement('span');
  tsEl.className = 'ts';
  tsEl.textContent = ts();

  // port badge (only on sent/recv lines)
  let badge = '';
  if (['sent','recv','ok','err','cme','imsi','qcom-resp'].includes(type)) {
    const pb = document.createElement('span');
    pb.className = 'portbadge p' + port;
    pb.textContent = 'P' + port;
    row.appendChild(pfx);
    row.appendChild(tsEl);
    row.appendChild(pb);
  } else if (['sent2','recv2'].includes(type)) {
    const pb = document.createElement('span');
    pb.className = 'portbadge p2';
    pb.textContent = 'P2';
    row.appendChild(pfx);
    row.appendChild(tsEl);
    row.appendChild(pb);
  } else {
    row.appendChild(pfx);
    row.appendChild(tsEl);
  }

  const txt = document.createElement('span');
  txt.className = 'text';
  txt.textContent = text;
  row.appendChild(txt);

  output.appendChild(row);
  output.scrollTop = output.scrollHeight;

  // accumulate P1 responses for bridge forward
  if (['recv','ok','imsi','qcom-resp'].includes(type) && port === 1) {
    state.lastP1Response = text;
  }
}

function clearOutput() {
  output.innerHTML = '';
  state.lineCount = 0;
  appendLine('output cleared', 'info');
}

/* ══════════════════════════════════════════════════════════════
   COMMAND HISTORY
═══════════════════════════════════════════════════════════════ */
function pushHistory(cmd, port) {
  // avoid duplicates at top
  if (state.cmdHistory.length && state.cmdHistory[0].cmd === cmd) return;
  state.cmdHistory.unshift({ cmd, port, time: ts() });
  if (state.cmdHistory.length > 200) state.cmdHistory.pop();
  state.histIdx = -1;
}

function renderHistory() {
  const el = document.getElementById('tab-history');
  if (!state.cmdHistory.length) {
    el.innerHTML = '<div style="color:var(--text3);padding:20px">No commands sent yet.</div>';
    return;
  }
  el.innerHTML = state.cmdHistory.map((h, i) =>
    `<div class="hist-row" onclick="recallHistory(${i})">
       <span class="hist-n">${i+1}</span>
       <span class="hist-port p${h.port}">P${h.port}</span>
       <span class="rk">${escHtml(h.cmd)}</span>
       <span style="margin-left:auto;color:#111e2a;font-size:10px">${h.time}</span>
     </div>`
  ).join('');
}

function recallHistory(i) {
  if (state.cmdHistory[i]) {
    cmdInput.value = state.cmdHistory[i].cmd;
    cmdInput.focus();
    switchTab('terminal', document.querySelector('.tab'));
    document.querySelectorAll('.tab')[0].classList.add('active');
  }
}

function escHtml(s) {
  return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

// classify a response line for colouring
function classifyResponse(line, port) {
  const u = line.toUpperCase().trim();
  if (u === 'OK')                          return 'ok';
  if (u === 'ERROR')                       return 'err';
  if (u.startsWith('+CME ERROR'))          return 'cme';
  if (u.startsWith('+CMS ERROR'))          return 'cme';
  if (u.startsWith('+CIMI') || /^\d{14,15}$/.test(u)) return 'imsi';
  if (u.startsWith('+CREG') || u.startsWith('+CEREG') || u.startsWith('+C5GREG')) return 'etsi-resp';
  if (u.startsWith('AT!') || u.startsWith('+CSQ'))    return 'qcom-resp';
  if (u.startsWith('AT%'))                             return 'qcom-resp';
  if (u.startsWith('+CSIM') || u.startsWith('+CGLA')) return 'apdu-resp';
  if (u.startsWith('+COPS') || u.startsWith('+CGDCONT')) return 'etsi-resp';
  if (port === 2)                          return 'recv2';
  return 'recv';
}

// core send — port 1 (S21)
async function sendCmd(cmd) {
  cmd = cmd.trim();
  if (!cmd) return;
  pushHistory(cmd, 1);
  appendLine(cmd, 'sent', 1);

  if (state.simMode || !state.writer1) {
    simulateResponse(cmd, 1);
    return;
  }
  try {
    const enc = new TextEncoder();
    await state.writer1.write(enc.encode(cmd + '\r\n'));
  } catch(e) {
    appendLine('Port 1 write error: ' + e.message, 'err', 1);
  }
}

// core send — port 2 (external modem)
async function sendCmdPort2(cmd) {
  cmd = cmd.trim();
  if (!cmd) return;
  pushHistory(cmd, 2);
  appendLine(cmd, 'sent2', 2);

  if (state.simMode || !state.writer2) {
    if (!state.port2 && !state.simMode) {
      appendLine('Port 2 not connected — connect it first or enable sim mode', 'err', 2);
      return;
    }
    simulateResponse(cmd, 2);
    return;
  }
  try {
    const enc = new TextEncoder();
    await state.writer2.write(enc.encode(cmd + '\r\n'));
  } catch(e) {
    appendLine('Port 2 write error: ' + e.message, 'err', 2);
  }
}

// send from typed input row (respects state.inputTarget)
function sendFromInput() {
  const cmd = cmdInput.value.trim();
  if (!cmd) return;
  cmdInput.value = '';
  state.histIdx = -1;
  if (state.inputTarget === 2) {
    sendCmdPort2(cmd);
  } else {
    sendCmd(cmd);
  }
}

// forward last P1 response as a command to P2
function forwardLastToP2() {
  if (!state.lastP1Response) {
    appendLine('No P1 response to forward yet', 'info');
    return;
  }
  appendLine('bridging: ' + state.lastP1Response + ' → Port 2', 'info');
  sendCmdPort2(state.lastP1Response);
}

/* ══════════════════════════════════════════════════════════════
   KEYBOARD — arrow history, Enter to send
═══════════════════════════════════════════════════════════════ */
cmdInput.addEventListener('keydown', e => {
  if (e.key === 'Enter') {
    e.preventDefault();
    sendFromInput();
  } else if (e.key === 'ArrowUp') {
    e.preventDefault();
    if (state.cmdHistory.length) {
      state.histIdx = Math.min(state.histIdx + 1, state.cmdHistory.length - 1);
      cmdInput.value = state.cmdHistory[state.histIdx].cmd;
    }
  } else if (e.key === 'ArrowDown') {
    e.preventDefault();
    state.histIdx = Math.max(state.histIdx - 1, -1);
    cmdInput.value = state.histIdx < 0 ? '' : state.cmdHistory[state.histIdx].cmd;
  }
});

/* ══════════════════════════════════════════════════════════════

═══════════════════════════════════════════════════════════════ */
p1Pill.addEventListener('click', () => {

  else connectPort(1);

p2Pill.addEventListener('click', () => {

function setPillState(portNum, connected, label) {
  const pill = portNum === 1 ? p1Pill : p2Pill;
  if (connected) {
    pill.textContent = label || (portNum === 1 ? 'S21 LIVE' : 'PORT2 LIVE');
    pill.style.background = portNum === 1 ? 'var(--green-dim)' : 'var(--p2dim)';
    pill.style.color       = portNum === 1 ? 'var(--green)'   : 'var(--p2color)';
    pill.style.borderColor = portNum === 1 ? 'var(--green)'   : 'var(--p2color)';
  } else {
    pill.textContent = portNum === 1 ? 'S21 OFFLINE' : 'PORT2 OFF';
    pill.style.background = '';
    pill.style.color = '';
    pill.style.borderColor = '';
  }
}

function showPanel(id) {
  const def = PANELS[id];
  if (!def) return;

  // special case: forward immediately
  if (id === 'fwd_p2') { forwardLastToP2(); return; }

  panelTitle.textContent = def.title;
  paramGrid.innerHTML = '';

  def.fields.forEach(f => {
    const wrap  = document.createElement('div');
    wrap.className = 'param-field';
    const lbl = document.createElement('label');
    lbl.className = 'param-label';
    lbl.textContent = f.label;
    const inp = document.createElement('input');
    inp.className   = 'param-input';
    inp.id          = 'pf_' + f.id;
    inp.placeholder = f.placeholder || '';
    inp.autocomplete = 'off';
    inp.spellcheck   = false;
    wrap.appendChild(lbl);
    wrap.appendChild(inp);
    paramGrid.appendChild(wrap);
  });

  paramPanel.classList.add('open');
  // focus first input
  const first = paramGrid.querySelector('.param-input');
  if (first) first.focus();

  // stash which panel is open
  paramPanel.dataset.panelId = id;
}

function closePanel() {
  paramPanel.classList.remove('open');
  paramPanel.dataset.panelId = '';
}

function buildFromPanel() {
  const id  = paramPanel.dataset.panelId;
  const def = PANELS[id];
  if (!def) return;

  const vals = {};
  def.fields.forEach(f => {
    const el = document.getElementById('pf_' + f.id);
    if (el) vals[f.id] = el.value.trim();
  });

  const targetPort = parseInt(panelPortSel.value) || 1;

  // special multi-cmd panels
  if (def.multi) {
    const cmds = def.multi(vals);
    cmds.forEach((c,i) => setTimeout(() => {
      targetPort === 2 ? sendCmdPort2(c) : sendCmd(c);
    }, i * 300));
    closePanel();
    return;
  }

  const cmd = def.build(vals);
  if (!cmd) { closePanel(); return; }

  if (targetPort === 2) sendCmdPort2(cmd);
  else sendCmd(cmd);
  closePanel();
}

/* ══════════════════════════════════════════════════════════════
   SILENT SMS PDU BUILDER (class 0)
═══════════════════════════════════════════════════════════════ */
function buildSilentPDU(smsc, dest, msgText) {
  // SMSC field — 00 means use default stored on SIM
  const smscHex = smsc ? encodePhoneNum(smsc) : '00';

  // Encode destination
  const destSemi = dest.replace('+','');
  const destType = dest.startsWith('+') ? '91' : '81';
  const destEncoded = semiOctetEncode(destSemi);
  const destLen = destSemi.length.toString(16).padStart(2,'0');

  // TP-OA not needed for SMS-SUBMIT
  // TP flags: 01 = SMS-SUBMIT, no VP, class 0 TP-DCS
  const tpFlags = '01';
  const mr      = '00';          // message reference
  const pid     = '00';          // protocol ID
  const dcs     = 'F0';          // class 0, GSM7 bit 7=1

  // Encode text as GSM-7 packed
  const gsm7 = textToGSM7Packed(msgText.slice(0,160));
  const udl  = msgText.slice(0,160).length.toString(16).padStart(2,'0');

  const tpdu = tpFlags + mr + destLen + destType + destEncoded + pid + dcs + udl + gsm7;
  const tpduLen = tpdu.length / 2;

  return {
    hex: smscHex + tpdu,
    tpduLen,
  };
}

function semiOctetEncode(numStr) {
  // pad to even, swap nibbles
  if (numStr.length % 2) numStr += 'F';
  let out = '';
  for (let i = 0; i < numStr.length; i += 2) {
    out += numStr[i+1] + numStr[i];
  }
  return out.toUpperCase();
}

function encodePhoneNum(num) {
  const digits = num.replace('+','');
  const type   = num.startsWith('+') ? '91' : '81';
  const encoded = semiOctetEncode(digits);
  const len = Math.ceil((encoded.length / 2) + 1);
  return len.toString(16).padStart(2,'0') + type + encoded;
}

// Minimal GSM-7 default alphabet encode → packed septets
const GSM7_CHARS =
  '@£$¥èéùìòÇ\nØø\rÅåΔ_ΦΓΛΩΠΨΣΘΞ\x1bÆæßÉ !"#¤%&\'()*+,-./0123456789:;<=>?' +
  '¡ABCDEFGHIJKLMNOPQRSTUVWXYZÄÖÑÜ§¿abcdefghijklmnopqrstuvwxyzäöñüà';

function textToGSM7Packed(text) {
  const septets = [];
  for (const ch of text) {
    let idx = GSM7_CHARS.indexOf(ch);
    if (idx < 0) idx = 32; // space fallback
    septets.push(idx & 0x7F);
  }
  // pack 7-bit septets into octets
  let bits = 0, bitCount = 0, out = '';
  for (const s of septets) {
    bits |= (s << bitCount);
    bitCount += 7;
    while (bitCount >= 8) {
      out += (bits & 0xFF).toString(16).padStart(2,'0');
      bits >>= 8;
      bitCount -= 8;
    }
  }
  if (bitCount > 0) out += (bits & 0xFF).toString(16).padStart(2,'0');
  return out.toUpperCase();
}

function simulateResponse(cmd, port) {
  const key = cmd.trim().toUpperCase().split('=')[0].split('?')[0];
  const fullKey = cmd.trim().toUpperCase();

  // look for full match first, then prefix
  let responses = SIM_RESPONSES[fullKey] || SIM_RESPONSES[key];

  if (!responses) {
    // generic fallback
    if (fullKey.startsWith('AT!') || fullKey.startsWith('AT%')) {
      responses = ['(sim) no canned response for Qualcomm/Shannon cmd', 'OK'];
    } else if (fullKey.startsWith('AT+CSIM') || fullKey.startsWith('AT+CGLA')) {
      responses = ['+CSIM: 4,"9000"', 'OK'];
    } else {
      responses = ['OK'];
    }
  }

  const delay = 120 + Math.random() * 80;
  setTimeout(() => {
    responses.forEach((r, i) => {
      setTimeout(() => {
        r.split('\r\n').forEach(line => {
          if (!line.trim()) return;
          const type = classifyResponse(line, port);
          appendLine(line, type, port);
        });
      }, i * 60);
    });
  }, delay);
}
